$ docker compose exec postgres psql ... -c "SELECT CASE WHEN source = 'usgs-fdsn' THEN 'Histórico 2025' ELSE 'Feed 30/10-01/11/2026' END AS conjunto,
    umbral, COUNT(*) AS terremotos, COUNT(*) FILTER (WHERE country_id = 'USA') AS eeuu, ... pct_eeuu, ... pct_sin_pais
  FROM evento_actual e CROSS JOIN (VALUES ('todas', -99.0), ('M>=2.5', 2.5), ('M>=4', 4.0)) AS u(umbral, minimo)
  WHERE source IN ('usgs-feed-all_day', 'usgs-fdsn') AND COALESCE(mag, -99) >= minimo GROUP BY 1, 2 ORDER BY 1 DESC, 3 DESC;"
